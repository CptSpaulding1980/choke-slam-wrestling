// Countdown Timer
function updateCountdown() {
  // Set target date — UPDATE THIS when S08 date is confirmed
  const target = new Date('2026-07-31T20:00:00');
  const now = new Date();
  const diff = target - now;
  
  const el = document.getElementById('s08-countdown');
  if (diff <= 0) {
    el.innerHTML = '🔥 SEASON 8 IS LIVE! 🔥';
    return;
  }
  
  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
  const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
  const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
  
  el.innerHTML = `⏳ ${days}d ${hours}h ${mins}m bis S08`;
}
updateCountdown();
